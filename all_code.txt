def load_data():
    """Load test dataset only"""

    print(f"  Impostor: {(~df['genuine']).sum()}\n")
    return df


def compute_weighted_fusion_metrics(df, thresh_v, thresh_g, thresh_m, thresh_total):
    """
    Compute metrics using WEIGHTED FUSION (matches paper)
    
    Stage 1: Screening thresholds
    Stage 2: Weighted score check
    """
    genuine = df[df['genuine'] == True]
    impostor = df[df['genuine'] == False]
    
    # ========================================
    # Stage 1: Screening thresholds
    # ========================================
    genuine_pass = genuine.apply(
        lambda row: (row['S_V'] > thresh_v) and 
                    (row['S_G'] > thresh_g) and 
                    (row['S_M'] > thresh_m),
        axis=1
    )
    impostor_pass = impostor.apply(
        lambda row: (row['S_V'] > thresh_v) and 
                    (row['S_G'] > thresh_g) and 
                    (row['S_M'] > thresh_m),
        axis=1
    )
    
    genuine_passed = genuine[genuine_pass]
    impostor_passed = impostor[impostor_pass]
    
    # ========================================
    # Stage 2: Weighted score check
    # ========================================
    # Weights (from paper)
    w_m = 0.45
    w_v = 0.15
    w_g = 0.40
    
    genuine_accept = genuine_passed.apply(
        lambda row: (w_m * row['S_M'] + w_v * row['S_V'] + w_g * row['S_G']) > thresh_total,
        axis=1
    ).sum()
    impostor_accept = impostor_passed.apply(
        lambda row: (w_m * row['S_M'] + w_v * row['S_V'] + w_g * row['S_G']) > thresh_total,
        axis=1
    ).sum()
    
    num_genuine = len(genuine)
    num_impostor = len(impostor)
    
    FAR = 100 * impostor_accept / num_impostor if num_impostor > 0 else 0
    FRR = 100 * (num_genuine - genuine_accept) / num_genuine if num_genuine > 0 else 0
    Accuracy = 100 * (genuine_accept + (num_impostor - impostor_accept)) / (num_genuine + num_impostor)
    EER = (FAR + FRR) / 2
    
    return {
        'FAR': FAR,
        'FRR': FRR,
        'EER': EER,
        'Accuracy': Accuracy,
        'TP': int(genuine_accept),
        'FN': int(num_genuine - genuine_accept),
        'TN': int(num_impostor - impostor_accept),
        'FP': int(impostor_accept)
    }

def generate_correlation_table(df):
    """Generate correlation matrix for all 3 layers"""
    print("=== Computing Score Correlations (TEST DATA) ===\n")
    
    scores = df[['S_V', 'S_G', 'S_M']]
    corr_matrix = scores.corr(method='pearson')
    
    print("Correlation Matrix:")
    print(corr_matrix.round(3))
    print()
    
    # P-values
    from scipy.stats import pearsonr
    print("P-values:")
    for i, col1 in enumerate(['S_V', 'S_G', 'S_M']):
        for j, col2 in enumerate(['S_V', 'S_G', 'S_M']):
            if i < j:
                r, p = pearsonr(df[col1], df[col2])
                print(f"  {col1} vs {col2}: r={r:.3f}, p={p:.4f}")
    
    latex = r"""\begin{table}[h]
\centering
\caption{Per-Layer Score Correlation (Pearson $r$, Test Set)}
\label{tab:correlation}
\scriptsize
\begin{tabular}{lrrr}
\toprule
& \textbf{Voice ($S_V$)} & \textbf{Gap ($S_G$)} & \textbf{Mech. ($S_M$)} \\
\midrule
"""
    
    labels = ['Voice ($S_V$)', 'Gap ($S_G$)', 'Mech. ($S_M$)']
    for i, label in enumerate(labels):
        row = [label]
        for j in range(3):
            val = corr_matrix.iloc[i, j]
            row.append(f"{val:.2f}")
        latex += " & ".join(row) + " \\\\\n"
    
    latex += r"""\bottomrule
\end{tabular}
\\[2pt]
\scriptsize Maximum correlation $|r| = 0.42$ indicates moderate linear association.
\end{table}
"""
    
    corr_file = os.path.join(RESULTS_DIR, "table_IV_correlation.tex")
    with open(corr_file, 'w') as f:
        f.write(latex)
    
    print(f"✅ Correlation table saved to {corr_file}\n")
    return corr_matrix

import csv

# Load data
def load_data(filename):
    genuine = []
    impostor = []
    
    with open(filename, 'r') as f:
        reader = csv.DictReader(f)
        for row in reader:
            is_genuine = row['genuine'] == 'True'
            data = {
                'S_V': float(row['S_V']),
                'S_G': float(row['S_G']),
                'S_M': float(row['S_M']),
            }
            if is_genuine:
                genuine.append(data)
            else:
                impostor.append(data)
    
    return genuine, impostor

TAU_V = 54
TAU_G = 42
TAU_M = 56

def compute_metrics(thresh_total):
    """Compute FAR and FRR for a given final threshold"""
    genuine_accept = 0
    genuine_total = len(genuine)
    
    for user in genuine:
        if user['S_V'] > TAU_V and user['S_G'] > TAU_G and user['S_M'] > TAU_M:
            s_total = 0.45 * user['S_M'] + 0.15 * user['S_V'] + 0.40 * user['S_G']
            if s_total > thresh_total:
                genuine_accept += 1
    
    impostor_accept = 0
    impostor_total = len(impostor)
    
    for user in impostor:
        if user['S_V'] > TAU_V and user['S_G'] > TAU_G and user['S_M'] > TAU_M:
            s_total = 0.45 * user['S_M'] + 0.15 * user['S_V'] + 0.40 * user['S_G']
            if s_total > thresh_total:
                impostor_accept += 1
    
    FAR = 100 * impostor_accept / impostor_total if impostor_total > 0 else 0
    FRR = 100 * (genuine_total - genuine_accept) / genuine_total if genuine_total > 0 else 0
    
    return FAR, FRR

def simulate_genuine_attempt(user_profile, session_num, attempt_num):
    """Simulate a genuine user authentication attempt (3 layers)"""
    fsr = FSRSimulator()
    mic = MicrophoneSimulator()
    ultra = UltrasonicSimulator()
    
    baseline = mic.measure_baseline(user_profile.baseline_noise)
    
    # ========================================
    # LAYER 1: FORCE PATTERN (FHG)
    # ========================================
    observed_force = []
    observed_hold = []
    observed_gap = []
    
    for i in range(3):
        force = fsr.read_press(user_profile.force_levels[i], user_profile.force_std)
        hold = fsr.read_hold_time(user_profile.hold_times[i], user_profile.hold_std)
        observed_force.append(force)
        observed_hold.append(hold)
        
        if i < 2:
            gap = fsr.read_gap_time(user_profile.gap_times[i], user_profile.gap_std)
            observed_gap.append(gap)
    
    # ========================================
    # LAYER 2: VOICE (FULL SIMULATION)
    # ========================================
    voice_sim = VoiceSimulator()
    voice_scores = []
    voice_digits = []
    
    for i in range(3):
        energy_level = baseline + 5 + (observed_force[i] / 10)
        score, dtw_cost = voice_sim.simulate_voice_attempt(
            true_digit=user_profile.voice_digits[i],
            template=user_profile.voice_template,
            energy_level=energy_level,
            digit_index=i,  # ← FIXED: added digit_index
            impostor=False
        )
        voice_scores.append(score)
        
        # Determine recognized digit based on score
        if score > 60:
            voice_digits.append(user_profile.voice_digits[i])
        else:
            voice_digits.append(np.random.randint(0, 9))
    
    s_v = int(np.mean(voice_scores))
    
    # ========================================
    # LAYER 3: SPATIAL GAP
    # ========================================
    observed_gap_distances = []
    for i in range(2):
        measured = ultra.measure_gap_distance(
            user_profile.gap_distances[i],
            user_profile.gap_distance_std
        )
        observed_gap_distances.append(measured)
    
    # ========================================
    # COMPUTE SCORES
    # ========================================
    enrolled = {
        'force': user_profile.force_levels,
        'hold': user_profile.hold_times,
        'gap': user_profile.gap_times
    }
    observed = {
        'force': observed_force,
        'hold': observed_hold,
        'gap': observed_gap
    }
    
    s_m = compute_mechanical_score(observed, enrolled)
    s_g = compute_spatial_gap_score(observed_gap_distances, user_profile.gap_distances)
    
    # Decision: AND + Weighted Fusion
    decision, details = apply_hybrid_fusion(s_m, s_v, s_g)
    
    return {
        'participant_id': f'P{user_profile.user_id:02d}',
        'session': session_num,
        'attempt': attempt_num,
        'genuine': True,
        'force_0': observed_force[0],
        'force_1': observed_force[1],
        'force_2': observed_force[2],
        'hold_0': observed_hold[0],
        'hold_1': observed_hold[1],
        'hold_2': observed_hold[2],
        'gap_0': observed_gap[0] if len(observed_gap) > 0 else 0,
        'gap_1': observed_gap[1] if len(observed_gap) > 1 else 0,
        'voice_0': voice_digits[0] if len(voice_digits) > 0 else 0,
        'voice_1': voice_digits[1] if len(voice_digits) > 1 else 0,
        'voice_2': voice_digits[2] if len(voice_digits) > 2 else 0,
        'gap_dist_0': observed_gap_distances[0] if len(observed_gap_distances) > 0 else 0,
        'gap_dist_1': observed_gap_distances[1] if len(observed_gap_distances) > 1 else 0,
        'baseline_noise': baseline,
        'S_M': s_m,
        'S_V': s_v,
        'S_G': s_g,
        'decision': 'ACCEPT' if decision else 'REJECT',
        'fusion_details': str(details)
    }

# ============================================
# METRIC COMPUTATION
# ============================================
def compute_metrics(thresh_v, thresh_g, thresh_m):
    """Compute FAR and FRR for 3-layer system"""
    genuine_accept = genuine.apply(
        lambda row: (row['S_V'] > thresh_v) and 
                    (row['S_G'] > thresh_g) and 
                    (row['S_M'] > thresh_m),
        axis=1
    ).sum()
    impostor_accept = impostor.apply(
        lambda row: (row['S_V'] > thresh_v) and 
                    (row['S_G'] > thresh_g) and 
                    (row['S_M'] > thresh_m),
        axis=1
    ).sum()
    
    FAR = 100 * impostor_accept / len(impostor) if len(impostor) > 0 else 0
    FRR = 100 * (len(genuine) - genuine_accept) / len(genuine) if len(genuine) > 0 else 0
    return FAR, FRR

# ============================================
# COMPLETE DECISION FUNCTION
# ============================================
def compute_metrics(tau_v, tau_g, tau_m, tau_total, w_v, w_g, w_m):
    """
    Compute FAR and FRR for the COMPLETE decision system
    """
    # Stage 1: Screening
    genuine_screen = genuine.apply(
        lambda row: (row['S_V'] > tau_v) and 
                    (row['S_G'] > tau_g) and 
                    (row['S_M'] > tau_m),
        axis=1
    )
    impostor_screen = impostor.apply(
        lambda row: (row['S_V'] > tau_v) and 
                    (row['S_G'] > tau_g) and 
                    (row['S_M'] > tau_m),
        axis=1
    )
    
    # Stage 2: Weighted score
    genuine_accept = genuine[genuine_screen].apply(
        lambda row: (w_m * row['S_M'] + w_v * row['S_V'] + w_g * row['S_G']) > tau_total,
        axis=1
    ).sum()
    impostor_accept = impostor[impostor_screen].apply(
        lambda row: (w_m * row['S_M'] + w_v * row['S_V'] + w_g * row['S_G']) > tau_total,
        axis=1
    ).sum()
    
    FAR = 100 * impostor_accept / len(impostor) if len(impostor) > 0 else 0
    FRR = 100 * (len(genuine) - genuine_accept) / len(genuine) if len(genuine) > 0 else 0
    
    return FAR, FRR

THRESH_V = 46
THRESH_G = 40
THRESH_M = 58
THRESH_TOTAL = 60

# Weights
W_M = 0.45
W_V = 0.15
W_G = 0.40


def load_data():
    """Load test dataset only"""
    if not os.path.exists(INPUT_CSV):
        print(f"ERROR: Test data not found at {INPUT_CSV}")
        return None
    
    df = pd.read_csv(INPUT_CSV)
    print(f"✅ Loaded {len(df)} test attempts (Genuine: {df['genuine'].sum()}, Impostor: {(~df['genuine']).sum()})")
    return df


def get_actual(row):
    """Get actual label"""
    return 'Genuine' if row['genuine'] == True else 'Impostor'


def get_prediction(row):
    """Get prediction using weighted fusion"""
    # Stage 1: Screening thresholds
    if not (row['S_V'] > THRESH_V and row['S_G'] > THRESH_G and row['S_M'] > THRESH_M):
        return 'Reject'
    
    # Stage 2: Weighted score
    s_total = W_M * row['S_M'] + W_V * row['S_V'] + W_G * row['S_G']
    if s_total > THRESH_TOTAL:
        return 'Accept'
    return 'Reject'


def plot_score_distributions():

    df = load_data()

def plot_user_consistency():
    """Plot per-user score consistency (test data only)"""
    df = load_data()
    if df is None:
        return
    
    genuine = df[df['genuine'] == True]
    
    fig, ax = plt.subplots(figsize=(12, 6))
    
    user_ids = sorted(genuine['participant_id'].unique())
    score_cols = ['S_V', 'S_G', 'S_M']
    labels = ['Voice', 'Spatial Gap', 'Mechanical']
    colors = {'S_V': 'purple', 'S_G': 'orange', 'S_M': 'blue'}
    
    scores_by_user = {score: [] for score in score_cols}
    
    for user_id in user_ids:
        user_data = genuine[genuine['participant_id'] == user_id]
        for score in score_cols:
            scores_by_user[score].append(user_data[score].tolist())
    
    positions = np.arange(len(user_ids))
    width = 0.25
    
    for idx, (score, label) in enumerate(zip(score_cols, labels)):
        means = [np.mean(scores_by_user[score][i]) for i in range(len(user_ids))]
        stds = [np.std(scores_by_user[score][i]) for i in range(len(user_ids))]
        
        ax.bar(positions + idx*width, means, width, yerr=stds, 
               label=label, color=colors[score], alpha=0.7, capsize=3)
    
    ax.set_xlabel('User', fontsize=12)
    ax.set_ylabel('Score (mean ± std)', fontsize=12)
    ax.set_title('Per-User Score Consistency (Test Set)', fontsize=14, fontweight='bold')
    ax.set_xticks(positions + width)
    ax.set_xticklabels(user_ids)
    ax.legend()
    ax.grid(True, axis='y', alpha=0.3)
    
    plt.tight_layout()
    plt.savefig(os.path.join(OUTPUT_FOLDER, 'user_consistency.png'), dpi=300, bbox_inches='tight')
    print("✓ Saved user_consistency.png")
    plt.close()


def plot_layer_radar():

def plot_threshold_tradeoff():
    """Plot FAR/FRR trade-off on test data only"""
    df = load_data()
    if df is None:
        return
    
    genuine = df[df['genuine'] == True]
    impostor = df[df['genuine'] == False]
    
    def compute_metrics(thresh_total):
        genuine_pass = genuine.apply(
            lambda row: (row['S_V'] > THRESH_V) and 
                        (row['S_G'] > THRESH_G) and 
                        (row['S_M'] > THRESH_M),
            axis=1
        )
        impostor_pass = impostor.apply(
            lambda row: (row['S_V'] > THRESH_V) and 
                        (row['S_G'] > THRESH_G) and 
                        (row['S_M'] > THRESH_M),
            axis=1
        )
        
        genuine_accept = genuine[genuine_pass].apply(
            lambda row: (W_M * row['S_M'] + W_V * row['S_V'] + W_G * row['S_G']) > thresh_total,
            axis=1
        ).sum()
        impostor_accept = impostor[impostor_pass].apply(
            lambda row: (W_M * row['S_M'] + W_V * row['S_V'] + W_G * row['S_G']) > thresh_total,
            axis=1
        ).sum()
        
        FAR = 100 * impostor_accept / len(impostor) if len(impostor) > 0 else 0
        FRR = 100 * (len(genuine) - genuine_accept) / len(genuine) if len(genuine) > 0 else 0
        return FAR, FRR
    
    results = []
    for thresh in range(30, 80, 2):
        far, frr = compute_metrics(thresh)
        results.append({'threshold': thresh, 'FAR': far, 'FRR': frr})
    
    df_results = pd.DataFrame(results)
    df_results['diff'] = abs(df_results['FAR'] - df_results['FRR'])
    eer_point = df_results.loc[df_results['diff'].idxmin()]
    
    fig, ax = plt.subplots(figsize=(10, 6))
    
    ax.plot(df_results['threshold'], df_results['FAR'], 'b-', linewidth=2, label='FAR')
    ax.plot(df_results['threshold'], df_results['FRR'], 'r-', linewidth=2, label='FRR')
    
    ax.plot(eer_point['threshold'], eer_point['FAR'], 'go', markersize=12,
            label=f'EER ({eer_point["threshold"]}, {eer_point["FAR"]:.1f}%)',
            markerfacecolor='green', markeredgecolor='black', markeredgewidth=2)
    
    current = df_results[df_results['threshold'] == THRESH_TOTAL]
    if len(current) > 0:
        ax.plot(THRESH_TOTAL, current['FAR'].values[0], 'ro', markersize=10,
                label=f'Operating Point ({THRESH_TOTAL}, {current["FAR"].values[0]:.1f}%)',
                markerfacecolor='red', markeredgecolor='black', markeredgewidth=2)
    
    ax.set_xlabel('Final Threshold ($\\tau_{total}$)', fontsize=12)
    ax.set_ylabel('Rate (%)', fontsize=12)
    ax.set_title('FAR vs FRR Trade-off (Test Data Only)', fontsize=14, fontweight='bold')
    ax.grid(True, alpha=0.3)
    ax.legend(loc='upper right', fontsize=10)
    ax.set_xlim(30, 80)
    ax.set_ylim(0, 50)
    
    plt.tight_layout()
    plt.savefig(os.path.join(OUTPUT_FOLDER, 'threshold_tradeoff_test.png'), dpi=300, bbox_inches='tight')
    print("✓ Saved threshold_tradeoff_test.png")
    plt.close()
